<script>
  function getStudentResult(){
    return new Promise((resolve, reject)=> {
      console.log("Requesting student result...");

        if (success){
          resolve({
          id:"23-53088-3",
          name:"Mysha",
          department:"CSE",
          marks:85
          });
          }

          reject("Failed to retrive student result");

      },3000 );

  async function displayResult()
  {

  console.log("Marks:", student.marks);
 
  }

  }
  }
